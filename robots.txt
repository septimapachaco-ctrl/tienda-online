function loadState(key, fallback) {
  try {
    const raw = localStorage.getItem(key);
    return raw ? JSON.parse(raw) : fallback;
  } catch {
    return fallback;
  }
}

function saveState(key, value) {
  localStorage.setItem(key, JSON.stringify(value));
}

const state = {
  view: 'home',
  selectedProductId: null,
  searchTerm: '',
  categoryFilter: 'Todos',
  adminTab: 'products',

  products: loadState(STORAGE_KEYS.products, initialProducts),
  orders: loadState(STORAGE_KEYS.orders, defaultOrders),
  clients: loadState(STORAGE_KEYS.clients, defaultClients),
  coupons: loadState(STORAGE_KEYS.coupons, defaultCoupons),
  selectedVariant: {},
  notice: ''
};

function formatPrice(value) {
  return new Intl.NumberFormat('es-AR', {
    style: 'currency',
    currency: 'ARS',
    maximumFractionDigits: 0
  }).format(value);
}

function saveAll() {
  saveState(STORAGE_KEYS.products, state.products);
  saveState(STORAGE_KEYS.cart, state.cart);
  saveState(STORAGE_KEYS.orders, state.orders);
  saveState(STORAGE_KEYS.clients, state.clients);
  saveState(STORAGE_KEYS.coupons, state.coupons);
}

function addToCart(productId, customVariant = null) {
  const product = state.products.find((item) => item.id === productId);
  if (!product) return;

  const defaultVariant = {
    color: product.colors[0],
    size: product.sizes[0]
  };

  const variant = customVariant || state.selectedVariant[productId] || defaultVariant;

  const itemIndex = state.cart.findIndex(
    (item) => item.productId === productId && item.color === variant.color && item.size === variant.size
  );

  if (itemIndex >= 0) {
    state.cart[itemIndex].qty += 1;
  } else {
    state.cart.push({
      productId,
      qty: 1,
      color: variant.color,
      size: variant.size
    });
  }

  state.notice = `${product.name} agregado al carrito.`;
  saveAll();
  render();
}

function updateQty(productId, color, size, delta) {
  const index = state.cart.findIndex(
    (item) => item.productId === productId && item.color === color && item.size === size
  );

  if (index === -1) return;

  state.cart[index].qty += delta;
  if (state.cart[index].qty <= 0) {
    state.cart.splice(index, 1);
  }

  saveAll();
  render();
}

function removeFromCart(productId, color, size) {
  state.cart = state.cart.filter(
    (item) => !(item.productId === productId && item.color === color && item.size === size)
  );
  saveAll();
  render();
}

function getCartItems() {
  return state.cart.map((cartItem) => {
    const product = state.products.find((item) => item.id === cartItem.productId);
    return {
      ...cartItem,
      product
    };
  }).filter((item) => item.product);
}

function subtotalCart() {
  return getCartItems().reduce((sum, item) => sum + item.product.price * item.qty, 0);
}

function cartCount() {
  return state.cart.reduce((sum, item) => sum + item.qty, 0);
}

function handleCheckout(event) {
  event.preventDefault();

  const formData = new FormData(event.target);
  const name = formData.get('name');
  const email = formData.get('email');
  const city = formData.get('city');
  const shipping = formData.get('shipping');
  const payment = formData.get('payment');

  if (!state.cart.length) {
    state.notice = 'Tu carrito está vacío.';
    render();
    return;
  }

  const orderId = `PED-${Date.now().toString().slice(-5)}`;
  const total = subtotalCart() + (shipping === 'retiro' ? 0 : 2200);

  const order = {
    id: orderId,
    customer: name,
    email,
    city,
    total,
    status: payment === 'mercadopago' ? 'pagado' : 'pendiente',
    date: new Date().toISOString().slice(0, 10),
    items: state.cart.reduce((sum, item) => sum + item.qty, 0),
    shipping,
    payment
  };

  state.orders.unshift(order);
  state.clients = state.clients.some((entry) => entry.email === email)
    ? state.clients
    : [{ name, email, city, role: 'Nuevo cliente' }, ...state.clients];

  getCartItems().forEach(({ product, qty, color, size }) => {
    const target = state.products.find((item) => item.id === product.id);
    if (target) {
      target.stock = Math.max(0, target.stock - qty);
    }
  });

  state.cart = [];
  state.notice = `Pedido ${orderId} registrado con éxito.`;
  saveAll();
  state.view = 'catalog';
  render();
}

function updateOrderStatus(orderId, nextStatus) {
  state.orders = state.orders.map((order) =>
    order.id === orderId ? { ...order, status: nextStatus } : order
  );
  saveAll();
  render();
}

function addProduct(event) {
  event.preventDefault();
  const formData = new FormData(event.target);
  const product = {
    id: Date.now(),
    name: formData.get('name'),
    category: formData.get('category'),
    price: Number(formData.get('price')),
    stock: Number(formData.get('stock')),
    description: formData.get('description') || 'Producto agregado desde el panel administrativo.',
    image: formData.get('image') || 'https://images.unsplash.com/photo-1521572267360-ee0c2909d518?auto=format&fit=crop&w=900&q=80',
    featured: false,
    colors: ['Unicolor'],
    sizes: ['Único']
  };

  if (!product.name) return;

  state.products.unshift(product);
  saveAll();
  event.target.reset();
  render();
}

  };

  const navButtons = document.querySelectorAll('.nav-button');
  navButtons.forEach((button) => {

    button.classList.toggle('active', isActive);
  });

}

function attachEvents() {
  document.addEventListener('click', (event) => {
    const button = event.target.closest('[data-action]');
    if (!button) return;

    const action = button.dataset.action;

    if (action === 'nav') {
      state.view = button.dataset.view;
      state.notice = '';
      render();
      return;
    }

    if (action === 'quick-add') {
      addToCart(Number(button.dataset.productId));
      return;
    }

    if (action === 'open-product') {
      state.selectedProductId = Number(button.dataset.productId);
      state.view = 'product';
      render();
      return;
    }

    if (action === 'select-color' || action === 'select-size') {
      const productId = Number(button.dataset.productId);
      if (!state.selectedVariant[productId]) {
        state.selectedVariant[productId] = {};
      }
      if (action === 'select-color') {
        state.selectedVariant[productId].color = button.dataset.color;
      }
      if (action === 'select-size') {
        state.selectedVariant[productId].size = button.dataset.size;
      }
      render();
      return;
    }

    if (action === 'add-selected-to-cart') {
      addToCart(Number(button.dataset.productId));
      return;
    }

    if (action === 'qty-change') {
      updateQty(Number(button.dataset.productId), button.dataset.color, button.dataset.size, Number(button.dataset.delta));
      return;
    }

    if (action === 'remove-item') {
      removeFromCart(Number(button.dataset.productId), button.dataset.color, button.dataset.size);
      return;
    }

    if (action === 'admin-tab') {
      state.adminTab = button.dataset.tab;
      render();
      return;
    }

    if (action === 'filter-category') {
      state.categoryFilter = button.dataset.category;
      render();
      return;
    }
  });

  document.addEventListener('input', (event) => {
    const target = event.target;
    if (target.dataset.action === 'search') {
      state.searchTerm = target.value.trim();
      render();
    }
  });

  document.addEventListener('submit', (event) => {
    if (event.target.id === 'checkout-form') {
      handleCheckout(event);
    }

    if (event.target.id === 'product-form') {
      addProduct(event);
    }
  });

  document.addEventListener('change', (event) => {
    if (event.target.matches('[data-action="update-status"]')) {
      updateOrderStatus(event.target.dataset.orderId, event.target.value);
    }
  });
}

attachEvents();
render();
